# versions runtime/win/x64/msys2-clang64.txt pins, with the SHA-256 of each package file:
# <package without the mingw-w64-clang-x86_64- prefix> <version> <sha256 of the .pkg.tar.zst>.

# (collect-dlls.sh). The compiler, the C runtime and the Windows headers come from llvm-mingw instead.
